-- Users
CREATE TABLE users (
    user_id INT AUTO_INCREMENT PRIMARY KEY,
    email VARCHAR(255) NOT NULL UNIQUE,
    nickname VARCHAR(50) NOT NULL UNIQUE, -- 추가된 닉네임
    password VARCHAR(255) NOT NULL,
    name VARCHAR(100) NOT NULL,
    phone VARCHAR(20),
    address VARCHAR(255),
    role ENUM('student','instructor','admin') DEFAULT 'student',
    birth_date DATE,
    gender ENUM('male','female'),
    email_verified BOOLEAN DEFAULT FALSE,
    status ENUM('active','suspended','deleted') DEFAULT 'active',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP
);

-- Instructor Profile
CREATE TABLE instructor_profile (
    instructor_id INT PRIMARY KEY,
    affiliation VARCHAR(255),
    bio TEXT,
    FOREIGN KEY (instructor_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- Instructor Certificate
CREATE TABLE instructor_certificate (
    certificate_id INT AUTO_INCREMENT PRIMARY KEY,
    instructor_id INT NULL,
    certificate_name VARCHAR(255),
    issuer VARCHAR(255),
    issued_date DATE,
    FOREIGN KEY (instructor_id) REFERENCES instructor_profile(instructor_id)
        ON DELETE SET NULL
);

-- Course
CREATE TABLE courses (
    course_id INT AUTO_INCREMENT PRIMARY KEY,
    instructor_id INT NULL,
    title VARCHAR(255),
    description TEXT,
    category VARCHAR(100),
    price DECIMAL(10,2) DEFAULT 0.00,
    is_free BOOLEAN DEFAULT FALSE,
    avg_rating DECIMAL(3,2) DEFAULT 0.0,
    status ENUM('draft','published','closed') DEFAULT 'draft',
    student_count INT DEFAULT 0,
    expiry_date DATE,
    live_limit INT DEFAULT 20,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    deleted_at TIMESTAMP NULL,
    FOREIGN KEY (instructor_id) REFERENCES instructor_profile(instructor_id)
        ON DELETE SET NULL
);

-- Course Materials
CREATE TABLE course_materials (
    material_id INT AUTO_INCREMENT PRIMARY KEY,
    course_id INT NULL,
    name VARCHAR(255),
    file_path VARCHAR(500),
    file_type VARCHAR(50),
    has_exam BOOLEAN DEFAULT FALSE,
    has_replay BOOLEAN DEFAULT TRUE,
    FOREIGN KEY (course_id) REFERENCES courses(course_id)
        ON DELETE SET NULL
);

-- Orders
CREATE TABLE orders (
    order_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NULL,
    total_amount DECIMAL(10,2),
    status ENUM('paid','pending','refund_requested','refunded') DEFAULT 'pending',
    payment_method VARCHAR(50),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    refunded_at TIMESTAMP NULL,
    refund_reason TEXT,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- Enrollment
CREATE TABLE enrollments (
    enrollment_id INT AUTO_INCREMENT PRIMARY KEY,
    order_id INT NULL,
    student_id INT NULL,
    course_id INT NULL,
    enrolled_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    expired_at DATE,
    FOREIGN KEY (order_id) REFERENCES orders(order_id)
        ON DELETE SET NULL,
    FOREIGN KEY (student_id) REFERENCES users(user_id)
        ON DELETE SET NULL,
    FOREIGN KEY (course_id) REFERENCES courses(course_id)
        ON DELETE SET NULL
);

-- Course Progress
CREATE TABLE course_progress (
    progress_id INT AUTO_INCREMENT PRIMARY KEY,
    enrollment_id INT NULL,
    course_id INT NULL,
    content_id INT,
    is_completed BOOLEAN DEFAULT FALSE,
    completed_at TIMESTAMP NULL,
    FOREIGN KEY (enrollment_id) REFERENCES enrollments(enrollment_id)
        ON DELETE SET NULL,
    FOREIGN KEY (course_id) REFERENCES courses(course_id)
        ON DELETE SET NULL
);

-- Cart
CREATE TABLE cart (
    cart_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NULL,
    course_id INT NULL,
    added_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE SET NULL,
    FOREIGN KEY (course_id) REFERENCES courses(course_id)
        ON DELETE SET NULL
);

-- Posts
CREATE TABLE posts (
    post_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NULL,
    title VARCHAR(255),
    content TEXT,
    category VARCHAR(100),
    views INT DEFAULT 0,
    likes INT DEFAULT 0,
    comments_count INT DEFAULT 0,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- Comments
CREATE TABLE comments (
    comment_id INT AUTO_INCREMENT PRIMARY KEY,
    post_id INT NULL,
    user_id INT NULL,
    content TEXT,
    likes INT DEFAULT 0,
    parent_comment_id INT NULL,
    is_deleted BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES posts(post_id)
        ON DELETE SET NULL,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE SET NULL,
    FOREIGN KEY (parent_comment_id) REFERENCES comments(comment_id)
        ON DELETE SET NULL
);

-- Post Attachments
CREATE TABLE post_attachments (
    file_id INT AUTO_INCREMENT PRIMARY KEY,
    post_id INT NULL,
    uploader_id INT NULL,
    file_name VARCHAR(255),
    file_path VARCHAR(500),
    file_type VARCHAR(50),
    file_size BIGINT,
    downloads INT DEFAULT 0,
    uploaded_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (post_id) REFERENCES posts(post_id)
        ON DELETE SET NULL,
    FOREIGN KEY (uploader_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- Notifications
CREATE TABLE notifications (
    notification_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NULL,
    content TEXT,
    type ENUM('payment','course','system'),
    is_read BOOLEAN DEFAULT FALSE,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- Messages
CREATE TABLE messages (
    message_id INT AUTO_INCREMENT PRIMARY KEY,
    sender_id INT NULL,
    receiver_id INT NULL,
    content TEXT,
    is_read BOOLEAN DEFAULT FALSE,
    sent_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (sender_id) REFERENCES users(user_id)
        ON DELETE SET NULL,
    FOREIGN KEY (receiver_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- User Logs
CREATE TABLE user_logs (
    log_id INT AUTO_INCREMENT PRIMARY KEY,
    user_id INT NULL,
    action VARCHAR(100),
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (user_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);

-- Books
CREATE TABLE books (
    book_id INT AUTO_INCREMENT PRIMARY KEY,
    instructor_id INT NULL,
    title VARCHAR(255),
    description TEXT,
    cover_image VARCHAR(500),
    FOREIGN KEY (instructor_id) REFERENCES instructor_profile(instructor_id)
        ON DELETE SET NULL
);

-- Mock Exams
CREATE TABLE mock_exams (
    exam_id INT AUTO_INCREMENT PRIMARY KEY,
    instructor_id INT NULL,
    student_id INT NULL,
    title VARCHAR(255),
    question TEXT,
    answer TEXT,
    score INT,
    taken_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (instructor_id) REFERENCES instructor_profile(instructor_id)
        ON DELETE SET NULL,
    FOREIGN KEY (student_id) REFERENCES users(user_id)
        ON DELETE SET NULL
);
